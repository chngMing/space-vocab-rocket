  <!-- 6.5 云端数据同步可选模块 (若环境提供 Firebase 配置则自动跨端持久化)    -->
  <!-- ===================================================================== -->
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
    import { getAuth, signInAnonymously, signInWithCustomToken } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
    import { getFirestore, doc, setDoc, deleteDoc, onSnapshot, collection } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

    (async function initCloudSyncIfConfigured() {
      try {
        if (typeof __firebase_config === "undefined" || !__firebase_config) return;
        const firebaseConfig = JSON.parse(__firebase_config);
        const appId = typeof __app_id !== "undefined" ? __app_id : "space-vocab-rocket";
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        if (typeof __initial_auth_token !== "undefined" && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }

        const user = auth.currentUser;
        if (!user) return;

        window.firestoreBridge = {
          ready: true,
          saveDoc: async (colName, id, data) => {
            if (!auth.currentUser) return;
            await setDoc(doc(db, "artifacts", appId, "public", "data", colName, id), data);
          },
          removeDoc: async (colName, id) => {
            if (!auth.currentUser) return;
            await deleteDoc(doc(db, "artifacts", appId, "public", "data", colName, id));
          }
        };

        // 实时监听公共词库、学生账号、派发任务与学习记录
        ["vocab_banks", "students", "tasks", "records"].forEach(colName => {
          const colRef = collection(db, "artifacts", appId, "public", "data", colName);
          onSnapshot(
            colRef,
            (snapshot) => {
              const items = [];
              snapshot.forEach(d => items.push(d.data()));
              if (typeof window.onCloudCollectionUpdated === "function") {
                window.onCloudCollectionUpdated(colName, items);
              }
            },
            (err) => {
              console.warn(`Firestore ${colName} listener notice:`, err.message);
            }
          );
        });
      } catch (e) {
        console.warn("Running in standalone memory mode:", e.message);
      }
    })();
  </script>

    // 规范化构建单个词条对象（自动提取括号同义拼写如 (also Earth)）
    function buildNormalizedVocabItem(rawWord, rawIpa = "", rawCn = "", extraVariants = [], selected = true) {
      let cleanWord = String(rawWord || "").trim().replace(/^\*+/, "").trim();
      const variants = new Set();

      // 提取括号内的别名，例如 "earth (also Earth)" -> word: "earth", variants: ["Earth"]
      const parenMatch = cleanWord.match(/\((?:also\s+)?([^)]+)\)/i);
      if (parenMatch) {
        const alt = parenMatch[1].trim();
        if (alt) variants.add(alt);
        cleanWord = cleanWord.replace(parenMatch[0], "").trim();
      }

      if (cleanWord) {
        variants.add(cleanWord);
      }
      (extraVariants || []).forEach(v => {
        const trimmed = String(v || "").trim();
        if (trimmed) variants.add(trimmed);
      });

      // 清理音标与中文中的页码尾巴（如 p. 52）
      let cleanIpa = String(rawIpa || "").trim();
      if (cleanIpa && !cleanIpa.startsWith("/") && !cleanIpa.startsWith("[")) {
        cleanIpa = `/${cleanIpa}/`;
      }
      const cleanCn = String(rawCn || "")
        .replace(/\bp\.\s*\d+\b/gi, "")
        .trim();

      return {
        id: Date.now() + Math.random(),
        word: cleanWord,
        accepts: Array.from(variants),
        ipa: cleanIpa,
        cn: cleanCn,
        selected
      };
    }

    const GEMINI_MODEL = "gemini-3-flash-preview";

    // 带指数退避重试的 Gemini API 调用封装
    async function callGeminiWithBackoff(payload, maxRetries = 3) {
      const apiKey = "";
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${apiKey}`;

      for (let attempt = 0; attempt <= maxRetries; attempt++) {
        try {
          const response = await fetch(apiUrl, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload)
          });

          if (!response.ok) {
            if ((response.status === 429 || response.status >= 500) && attempt < maxRetries) {
              await new Promise(res => setTimeout(res, Math.pow(2, attempt) * 1000));
              continue;
            }
            throw new Error(`API 请求失败 (${response.status})`);
          }

          const result = await response.json();
          const text = result?.candidates?.[0]?.content?.parts?.[0]?.text;
          if (!text) {
            throw new Error("未获取到有效的识别结果");
          }
          return JSON.parse(text);
        } catch (err) {
          if (attempt < maxRetries) {
            await new Promise(res => setTimeout(res, Math.pow(2, attempt) * 1000));
            continue;
          }
          throw err;
        }
      }
    }

    // 通用词表 JSON Schema
    const VOCAB_ARRAY_SCHEMA = {
      type: "ARRAY",
      items: {
        type: "OBJECT",
        properties: {
          word: { type: "STRING" },
          variants: { type: "ARRAY", items: { type: "STRING" } },
          phonetic: { type: "STRING" },
          meaning: { type: "STRING" }
        },
        required: ["word", "phonetic", "meaning"],
        propertyOrdering: ["word", "variants", "phonetic", "meaning"]
      }
    };

    // 1. 使用 Gemini API 从图片或 PDF 提取中英文与音标
    async function extractVocabFromBase64File(base64Data, mimeType) {
      const prompt =
        "请仔细识别这张图片/文档中的所有英语单词表条目。提取每个条目的：1) 英文单词或短语（word，去掉前面的星号 *）；2) 括号里的可接受变体拼写（variants，例如 '(also Earth)' 提取为 ['Earth']，'(also the Moon)' 提取为 ['the Moon']）；3) 国际音标（phonetic，保留斜杠如 /'plænɪt/，若原图缺失可为空字符串）；4) 中文释义（meaning，去掉右侧页码如 'p. 52'）。严格按列表顺序输出 JSON 数组。";

      const payload = {
        contents: [
          {
            role: "user",
            parts: [
              { text: prompt },
              {
                inlineData: {
                  mimeType: mimeType,
                  data: base64Data
                }
              }
            ]
          }
        ],
        generationConfig: {
          responseMimeType: "application/json",
          responseSchema: VOCAB_ARRAY_SCHEMA
        }
      };

      const extracted = await callGeminiWithBackoff(payload);
      return (Array.isArray(extracted) ? extracted : [])
        .filter(entry => entry && entry.word && entry.word.trim())
        .map(entry =>
          buildNormalizedVocabItem(entry.word, entry.phonetic, entry.meaning, entry.variants, true)
        );
    }

    // 3. AI 一键补全当前词表中缺失的音标或中文释义
    async function fillMissingVocabFieldsWithAI() {
      const missingIndices = [];
      const wordsToQuery = [];

      vocabManagerState.items.forEach((item, idx) => {
        if (item.word.trim() && (!item.ipa.trim() || !item.cn.trim())) {
          missingIndices.push(idx);
          wordsToQuery.push(item.word.trim());
        }
      });

      if (wordsToQuery.length === 0) {
        setImportStatus("success", "✅ 当前所有单词均已包含完整音标与中文释义，无需补全！", false);
        return;
      }

      setImportStatus("loading", `✨ AI 正在为 ${wordsToQuery.length} 个单词补全音标与中文释义...`, true);
      try {
        const prompt = `请为以下英语单词列表提供准确的国际音标（phonetic，带斜杠如 /'plænɪt/）和简明准确的中文教材释义（meaning）：\n${JSON.stringify(wordsToQuery)}`;
        const payload = {
          contents: [{ role: "user", parts: [{ text: prompt }] }],
          generationConfig: {
            responseMimeType: "application/json",
            responseSchema: VOCAB_ARRAY_SCHEMA
          }
        };

        const results = await callGeminiWithBackoff(payload);
        const resultMap = new Map();
        (results || []).forEach(r => {
          if (r && r.word) {
            resultMap.set(r.word.trim().toLowerCase(), r);
          }
        });

        let updatedCount = 0;
        missingIndices.forEach(idx => {
          const item = vocabManagerState.items[idx];
          const found = resultMap.get(item.word.trim().toLowerCase());
          if (found) {
            if (!item.ipa.trim() && found.phonetic) {
              item.ipa = found.phonetic.startsWith("/") ? found.phonetic : `/${found.phonetic}/`;
            }
            if (!item.cn.trim() && found.meaning) {
              item.cn = found.meaning.trim();
            }
            updatedCount++;
          }
        });

        renderVocabManagerTable();
        setImportStatus("success", `🎉 已成功为 ${updatedCount} 个单词补全音标与中文释义！`, false);
      } catch (err) {
        setImportStatus("error", `❌ AI 补全失败：${err.message || "网络异常，请稍后重试"}`, false);
      }
    }

    // 4. 本地智能正则文本解析器（支持 "word /ipa/ meaning"、CSV/TSV、纯英文列表）
    function parseVocabTextLocally(rawText) {
      const lines = String(rawText || "")
        .split(/\r?\n/)
        .map(l => l.trim())
        .filter(Boolean);

      const parsed = [];
      for (const line of lines) {
        // 跳过纯表头行
        if (/^(word|english|单词|英文)[\t,|]/i.test(line)) continue;

        // 情况 A: 包含标准音标 /.../ 或 [...]
        const ipaMatch = line.match(/(\/[^/]+\/|\[[^\]]+\])/);
        if (ipaMatch) {
          const ipa = ipaMatch[1];
          const splitIdx = line.indexOf(ipa);
          const leftWord = line.slice(0, splitIdx).trim();
          const rightMeaning = line.slice(splitIdx + ipa.length).trim();
          if (leftWord) {
            parsed.push(buildNormalizedVocabItem(leftWord, ipa, rightMeaning, [], true));
            continue;
          }
        }

        // 情况 B: Tab 或逗号分隔的多列格式
        const delimiter = line.includes("\t") ? "\t" : (line.includes(",") ? "," : null);
        if (delimiter) {
          const cols = line.split(delimiter).map(c => c.replace(/^["']|["']$/g, "").trim());
          if (cols[0]) {
            if (cols.length >= 3) {
              parsed.push(buildNormalizedVocabItem(cols[0], cols[1], cols.slice(2).join("；"), [], true));
            } else {
              parsed.push(buildNormalizedVocabItem(cols[0], "", cols[1] || "", [], true));
            }
            continue;
          }
        }

        // 情况 C: 左侧英文短语 + 右侧中文字符
        const cnMatch = line.match(/([\u4e00-\u9fa5（(].*)$/);
        if (cnMatch && cnMatch.index > 0) {
          const leftWord = line.slice(0, cnMatch.index).trim();
          const rightMeaning = cnMatch[1].trim();
          if (leftWord) {
            parsed.push(buildNormalizedVocabItem(leftWord, "", rightMeaning, [], true));
            continue;
          }
        }

        // 情况 D: 纯英文单词或短语
        if (/^[a-zA-Z\s'-]+$/.test(line)) {
          parsed.push(buildNormalizedVocabItem(line, "", "", [], true));
        }
      }
      return parsed;
    }

      const spinner = document.getElementById("importStatusSpinner");

        "bg-rose-50", "text-rose-700", "border-rose-200"
      );
      banner.classList.add("border");

      if (type === "loading") {
        banner.classList.add("bg-indigo-50", "text-indigo-700", "border-indigo-200");
      } else if (type === "success") {
        banner.classList.add("bg-emerald-50", "text-emerald-700", "border-emerald-200");

      }

        spinner.classList.remove("hidden");
      } else {
        spinner.classList.add("hidden");
      }
      text.textContent = message;
    }

    async function handleImageFileUpload(file) {
      if (!file) return;
      if (!file.type.startsWith("image/")) {
        setImportStatus("error", "⚠️ 请上传有效的图片文件（JPG、PNG、WEBP 等）。", false);
        return;
      }

      setImportStatus("loading", `🤖 Gemini AI 正在识别图片「${file.name}」中的单词、音标与中文...`, true);
      try {
        const base64Data = await fileToBase64(file);
        const items = await extractVocabFromBase64File(base64Data, file.type);
        const bankTitle = file.name.replace(/\.[^.]+$/, "");
        applyImportedItems(items, `图片识别：${bankTitle}`);
      } catch (err) {
        setImportStatus("error", `❌ 图片识别失败：${err.message || "请检查网络或更换清晰图片"}`, false);
      }
    }

    // 处理多格式文档上传 (.txt, .csv, .tsv, .json, .xlsx, .xls, .docx, .pdf)
    async function handleDocumentFileUpload(file) {
      if (!file) return;
      const fileName = file.name;
      const ext = fileName.split(".").pop().toLowerCase();
      const bankTitle = fileName.replace(/\.[^.]+$/, "");

      setImportStatus("loading", `📄 正在解析文档「${fileName}」...`, true);
      try {
        if (ext === "pdf") {
          const base64Data = await fileToBase64(file);
          const items = await extractVocabFromBase64File(base64Data, "application/pdf");
          applyImportedItems(items, `PDF词表：${bankTitle}`);
          return;
        }

        if (ext === "xlsx" || ext === "xls") {
          if (typeof XLSX === "undefined") {
            throw new Error("Excel 解析组件尚未加载完成，请稍后再试");
          }
          const arrayBuf = await file.arrayBuffer();
          const workbook = XLSX.read(arrayBuf, { type: "array" });
          const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
          const rows = XLSX.utils.sheet_to_json(firstSheet, { header: 1 });
          const items = [];
          rows.forEach((row, idx) => {
            if (!Array.isArray(row) || row.length === 0) return;
            const c0 = String(row[0] || "").trim();
            if (!c0 || (idx === 0 && /^(word|english|单词|英文|序号|#)$/i.test(c0))) return;
            // 若第一列是纯数字序号，则往后顺移一列
            if (/^\d+$/.test(c0) && row[1]) {
              items.push(buildNormalizedVocabItem(row[1], row[2] || "", row[3] || "", [], true));
            } else if (row.length >= 3) {
              items.push(buildNormalizedVocabItem(c0, row[1] || "", row[2] || "", [], true));
            } else {
              items.push(buildNormalizedVocabItem(c0, "", row[1] || "", [], true));
            }
          });
          applyImportedItems(items, `表格词表：${bankTitle}`);
          return;
        }

        if (ext === "docx") {
          if (typeof mammoth === "undefined") {
            throw new Error("Word 解析组件尚未加载完成，请稍后再试");
          }
          const arrayBuf = await file.arrayBuffer();
          const result = await mammoth.extractRawText({ arrayBuffer: arrayBuf });
          let items = parseVocabTextLocally(result.value);
          if (items.length === 0 && result.value.trim()) {
            items = await extractVocabFromTextWithAI(result.value);
          }
          applyImportedItems(items, `Word词表：${bankTitle}`);
          return;
        }

        if (ext === "json") {
          const text = await file.text();
          const json = JSON.parse(text);
          const arr = Array.isArray(json) ? json : (json.words || json.items || []);
          const items = arr
            .filter(entry => entry && (entry.word || typeof entry === "string"))
            .map(entry => {
              if (typeof entry === "string") return buildNormalizedVocabItem(entry);
              return buildNormalizedVocabItem(
                entry.word,
                entry.ipa || entry.phonetic || "",
                entry.cn || entry.meaning || "",
                entry.accepts || entry.variants || [],
                true
              );
            });
          applyImportedItems(items, `JSON词表：${bankTitle}`);
          return;
        }

        // .txt / .csv / .tsv 等纯文本格式
        const plainText = await file.text();
        let items = parseVocabTextLocally(plainText);
        if (items.length === 0 && plainText.trim()) {
          items = await extractVocabFromTextWithAI(plainText);
        }
        applyImportedItems(items, `文档词表：${bankTitle}`);
      } catch (err) {
        setImportStatus("error", `❌ 文档解析失败：${err.message || "文件格式不支持或内容为空"}`, false);
      }
    }

    function fileToBase64(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => {
          const res = String(reader.result || "");
          const base64 = res.includes(",") ? res.split(",")[1] : res;
          resolve(base64);
        };
        reader.onerror = () => reject(new Error("读取本地文件失败"));
        reader.readAsDataURL(file);
      });
    }

    /**
     * =========================================================================
     * 模块 B: Web Audio 合成音效引擎 (无外部音频依赖，即时合成火箭推进/警报/爆炸/胜利音效)
     * =========================================================================
     */
    class RocketSoundEngine {
      constructor() {
        this.ctx = null;
        this.isMuted = false;
      }

      ensureContext() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (AudioCtx) {
            this.ctx = new AudioCtx();
          }
        }
        if (this.ctx && this.ctx.state === "suspended") {
          this.ctx.resume().catch(() => {});
        }
      }

      toggleMute() {
        this.isMuted = !this.isMuted;
        return this.isMuted;
      }

      // 拼写正确：强力推进器点火升空音效
      playBoostThrust() {
        if (this.isMuted) return;
        this.ensureContext();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;

        // 1. 引擎能量拉升音调
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = "triangle";
        osc.frequency.setValueAtTime(160, now);
        osc.frequency.exponentialRampToValueAtTime(520, now + 0.38);
        osc.frequency.exponentialRampToValueAtTime(760, now + 0.72);

        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.22, now + 0.06);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.78);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.8);

        // 2. 喷射气流白噪声
        const bufferSize = Math.floor(this.ctx.sampleRate * 0.65);
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = noiseBuffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = noiseBuffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = "bandpass";
        filter.frequency.setValueAtTime(420, now);
        filter.frequency.exponentialRampToValueAtTime(1200, now + 0.65);
        filter.Q.value = 1.2;

        const noiseGain = this.ctx.createGain();
        noiseGain.gain.setValueAtTime(0.001, now);
        noiseGain.gain.linearRampToValueAtTime(0.18, now + 0.05);
        noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.65);

        noise.connect(filter);
        filter.connect(noiseGain);
        noiseGain.connect(this.ctx.destination);
        noise.start(now);
        noise.stop(now + 0.66);
      }

      // 低燃料最后 3 秒提醒滴答声
      playLowFuelTick() {
        if (this.isMuted) return;
        this.ensureContext();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(880, now);
        osc.frequency.exponentialRampToValueAtTime(620, now + 0.11);

        gain.gain.setValueAtTime(0.12, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.13);
      }

      // 拼写错误或超时：火箭坠毁爆炸音效
      playCrashExplosion() {
        if (this.isMuted) return;
        this.ensureContext();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;

        // 1. 低频重击轰鸣
        const subOsc = this.ctx.createOscillator();
        const subGain = this.ctx.createGain();
        subOsc.type = "sawtooth";
        subOsc.frequency.setValueAtTime(130, now);
        subOsc.frequency.exponentialRampToValueAtTime(28, now + 0.9);

        subGain.gain.setValueAtTime(0.28, now);
        subGain.gain.exponentialRampToValueAtTime(0.001, now + 0.95);

        subOsc.connect(subGain);
        subGain.connect(this.ctx.destination);
        subOsc.start(now);
        subOsc.stop(now + 0.96);

        // 2. 爆炸碎片气浪噪声
        const bufferSize = Math.floor(this.ctx.sampleRate * 1.1);
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = noiseBuffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = noiseBuffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = "lowpass";
        filter.frequency.setValueAtTime(950, now);
        filter.frequency.exponentialRampToValueAtTime(140, now + 1.05);

        const noiseGain = this.ctx.createGain();
        noiseGain.gain.setValueAtTime(0.32, now);
        noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 1.08);

        noise.connect(filter);
        filter.connect(noiseGain);
        noiseGain.connect(this.ctx.destination);
        noise.start(now);
        noise.stop(now + 1.1);
      }

      // 全部通关：胜利凯旋和弦音效
      playVictoryFanfare() {
        if (this.isMuted) return;
        this.ensureContext();
        if (!this.ctx) return;

        const now = this.ctx.currentTime;
        const notes = [523.25, 659.25, 783.99, 1046.5]; // C5 - E5 - G5 - C6
        notes.forEach((freq, idx) => {
          const start = now + idx * 0.11;
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = "triangle";
          osc.frequency.setValueAtTime(freq, start);

          gain.gain.setValueAtTime(0.001, start);
          gain.gain.linearRampToValueAtTime(0.18, start + 0.03);
          gain.gain.exponentialRampToValueAtTime(0.001, start + 0.55);

          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start(start);
          osc.stop(start + 0.58);
        });
      }
    }

    /**
     * =========================================================================
     * 模块 C: Web Speech API 英语单词语音合成朗读器 (支持美音/英音切换)
     * =========================================================================
     */
    class SpeechPronouncer {
      constructor(onStateChange) {
        this.accent = "en-US";
        this.onStateChange = onStateChange || (() => {});
        this.voices = [];
        this.fallbackTimer = null;

        if ("speechSynthesis" in window) {
          const loadVoices = () => {
            this.voices = window.speechSynthesis.getVoices() || [];
          };
          loadVoices();
          window.speechSynthesis.onvoiceschanged = loadVoices;
        }
      }

      setAccent(accent) {
        this.accent = accent;
      }

      pickVoice() {
        if (!this.voices.length && "speechSynthesis" in window) {
          this.voices = window.speechSynthesis.getVoices() || [];
        }
        const targetLower = this.accent.toLowerCase();
        return (
          this.voices.find(v => v.lang && v.lang.toLowerCase() === targetLower) ||
          this.voices.find(v => v.lang && v.lang.toLowerCase().startsWith(targetLower.slice(0, 2))) ||
          null
        );
      }

      speak(text) {
        if (!text || !("speechSynthesis" in window)) return;
        this.stop();

        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = this.accent;
        utterance.rate = 0.9;
        utterance.pitch = 1.0;

        const chosenVoice = this.pickVoice();
        if (chosenVoice) {
          utterance.voice = chosenVoice;
        }

        utterance.onstart = () => {
          this.onStateChange(true);
        };
        utterance.onend = () => {
          clearTimeout(this.fallbackTimer);
          this.onStateChange(false);
        };
        utterance.onerror = () => {
          clearTimeout(this.fallbackTimer);
          this.onStateChange(false);
        };

        this.onStateChange(true);
        window.speechSynthesis.speak(utterance);

        // 兜底定时器：防止部分浏览器未触发 onend 导致声波动画常亮
        this.fallbackTimer = setTimeout(() => {
          this.onStateChange(false);
        }, 2500);
      }

      stop() {
        clearTimeout(this.fallbackTimer);
        if ("speechSynthesis" in window) {
          window.speechSynthesis.cancel();
        }
        this.onStateChange(false);
      }
    }

    function spawnThrustParticles(x, y, angle, count, powerScale) {
      for (let i = 0; i < count; i++) {

        particles.push({

    /**
     * =========================================================================
     * 模块 F: 游戏流程控制 (开始、提交检验、冲刺、坠机、结算)
     * =========================================================================

    function shuffleArray(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    /**
     * =========================================================================
     * 模块 F2: AI 个性化学习诊断与建议生成引擎 (统计拼写vs超时、难度归因、易错词)
     * =========================================================================

    function buildRuleBasedFallbackGuidance(record, studentAllRecords) {
      const beginnerErrs = studentAllRecords
        .filter(r => r.difficulty === "beginner")
        .reduce((acc, r) => acc + (r.missedItems ? r.missedItems.length : 0), 0);
      const intermediateErrs = studentAllRecords
        .filter(r => r.difficulty === "intermediate")
        .reduce((acc, r) => acc + (r.missedItems ? r.missedItems.length : 0), 0);
      const totalSpelling = studentAllRecords.reduce((acc, r) => acc + (r.spellingErrCount || 0), 0);
      const totalTimeout = studentAllRecords.reduce((acc, r) => acc + (r.timeoutErrCount || 0), 0);

      const overallSummary = record.isSuccess
        ? `🎉 表现非常出色！本次「${record.difficultyName}」正确率高达 ${record.accuracyPct}%，成功赢得 1 朵小红花 🌺！`
        : `💪 本次「${record.difficultyName}」答对 ${record.correctCount}/${record.totalWords} 题（正确率 ${record.accuracyPct}%），已为你精准定位薄弱环节：`;

      let errorTypeAnalysis = "";
      if (totalSpelling === 0 && totalTimeout === 0) {
        errorTypeAnalysis = "零失误通关！无论是单词拼写准确度还是限时反应速度都非常稳定。";
      } else if (totalTimeout > totalSpelling) {
        errorTypeAnalysis = `累计统计显示「思考超时 (${totalTimeout}次)」多于「拼写错误 (${totalSpelling}次)」，说明对词义或读音有一定印象，但提取反应速度偏慢，需要加强限时听写熟练度。`;
      } else {
        errorTypeAnalysis = `累计统计显示「拼写错误 (${totalSpelling}次)」多于「思考超时 (${totalTimeout}次)」，说明反应速度达标，但字母组合细节（如双写字母、元音字母辨析）容易写错。`;
      }

      let difficultyDiagnosis = "";
      if (beginnerErrs > intermediateErrs && beginnerErrs > 0) {
        difficultyDiagnosis = `初级模式累计错题 (${beginnerErrs}题) 较多：初级提供中文提示却仍出错，说明对「中文释义与英文拼写」的语义对应记忆不够深，建议朗读中文词义后默写英文。`;
      } else if (intermediateErrs >= beginnerErrs && intermediateErrs > 0) {
        difficultyDiagnosis = `中级模式累计错题 (${intermediateErrs}题) 较多：中级提供国际音标提示仍出错，说明对「音标拼读与字母组合发音规律」掌握得不够扎实，建议对照音标逐音节拼读。`;
      } else if (record.difficulty === "ultimate" && !record.isSuccess) {
        difficultyDiagnosis = "终极难度无文字提示且仅限 10 秒，对听力辨音与键盘盲打速度要求极高，建议先在中级音标模式下巩固发音规律后再冲刺终极难度。";
      } else {
        difficultyDiagnosis = "各难度层级表现均衡，建议逐步向「中级音标模式」和「终极 10 秒盲听模式」发起更高阶挑战！";
      }

      const missedWords = (record.missedItems || []).map(m => m.word);
      const hasPhrase = missedWords.some(w => w.includes(" "));
      const hasLongWord = missedWords.some(w => w.length >= 7);
      let wordCategoryAnalysis = "";
      if (missedWords.length === 0) {
        wordCategoryAnalysis = "本次所有词汇均一次性拼写正确！可尝试导入更长难度的多音节词汇或短语词组继续进阶。";
      } else {
        const tags = [];
        if (hasPhrase) tags.push("双词组合短语（如空格分隔词组）");
        if (hasLongWord) tags.push("7字母以上多音节长难词");
        if (tags.length === 0) tags.push("基础核心高频词汇");
        wordCategoryAnalysis = `本次易错词集中于【${tags.join("、")}】（错词：${missedWords.join(", ")}）。建议点击上方复盘列表的小喇叭反复跟读 3 遍，按音节拆分记忆！`;
      }

      return {
        overallSummary,
        errorTypeAnalysis,
        difficultyDiagnosis,
        wordCategoryAnalysis
      };
    }

    async function generateAndAttachAiGuidance(record) {
      const loadingEl = document.getElementById("aiFeedbackLoading");
      const contentEl = document.getElementById("aiFeedbackContent");
      loadingEl.classList.remove("hidden");
      contentEl.classList.add("hidden");

      const studentAllRecords = portalDataStore.records.filter(r => r.studentId === record.studentId);
      const fallbackGuidance = buildRuleBasedFallbackGuidance(record, studentAllRecords);

      // 汇总跨次统计供 Gemini 深度诊断
      const statsSummary = {
        currentAttempt: {
          difficulty: record.difficultyName,
          accuracyPct: record.accuracyPct,
          isSuccess: record.isSuccess,
          spellingErrors: record.spellingErrCount,
          timeoutErrors: record.timeoutErrCount,
          missedWords: record.missedItems.map(m => ({
            word: m.word,
            ipa: m.ipa,
            cn: m.cn,
            studentTyped: m.userAnswer,
            reason: m.reason
          }))
        },
        cumulativeStatsAcrossSessions: {
          totalSessions: studentAllRecords.length,
          totalSpellingErrors: studentAllRecords.reduce((s, r) => s + (r.spellingErrCount || 0), 0),
          totalTimeoutErrors: studentAllRecords.reduce((s, r) => s + (r.timeoutErrCount || 0), 0),
          beginnerWrongCount: studentAllRecords
            .filter(r => r.difficulty === "beginner")
            .reduce((s, r) => s + (r.missedItems?.length || 0), 0),
          intermediateWrongCount: studentAllRecords
            .filter(r => r.difficulty === "intermediate")
            .reduce((s, r) => s + (r.missedItems?.length || 0), 0),
          ultimateWrongCount: studentAllRecords
            .filter(r => r.difficulty === "ultimate")
            .reduce((s, r) => s + (r.missedItems?.length || 0), 0),
          historicalMissedWords: studentAllRecords
            .flatMap(r => (r.missedItems || []).map(m => m.word))
            .slice(0, 25)
        }
      };

      let finalGuidance = fallbackGuidance;
      try {
        const langDirective = userPrefs.uiLang === "en"
          ? "Please write all 4 JSON fields in clear, encouraging English."
          : "请使用温暖、专业的简体中文撰写全部 4 个 JSON 字段。";
        const prompt = `你是一位温暖、专业的少儿英语听写与自然拼读 AI 导师。${langDirective} 请根据以下学生的听写数据（包含本次与历史累计统计），生成既鼓励学生、又方便家长查看学情的个性化诊断报告：
1. overallSummary: 一句话总结本次表现与鼓励（若通关请祝贺获得小红花）。
2. errorTypeAnalysis: 统计并分析学生是「拼写错误」多还是「思考超时」多，说明原因。
3. difficultyDiagnosis: 对比初级（中文提示）、中级（音标提示）、终极（无提示）的错误情况——如果初级错误多，明确指出对中文词义与英文的对应记忆不深；如果中级错误多，明确指出国际音标与自然拼读掌握得不够扎实；如果终极错误多，给出盲听反应训练建议。
4. wordCategoryAnalysis: 分析学生一般比较容易错哪类单词（如多音节词、带空格短语、特定元音字母组合等），并给出具体可执行的记忆技巧。

学生数据 JSON：
${JSON.stringify(statsSummary)}`;

        const payload = {
          contents: [{ role: "user", parts: [{ text: prompt }] }],
          generationConfig: {
            responseMimeType: "application/json",
            responseSchema: {
              type: "OBJECT",
              properties: {
                overallSummary: { type: "STRING" },
                errorTypeAnalysis: { type: "STRING" },
                difficultyDiagnosis: { type: "STRING" },
                wordCategoryAnalysis: { type: "STRING" }
              },
              required: ["overallSummary", "errorTypeAnalysis", "difficultyDiagnosis", "wordCategoryAnalysis"],
              propertyOrdering: ["overallSummary", "errorTypeAnalysis", "difficultyDiagnosis", "wordCategoryAnalysis"]
            }
          }
        };

        const aiResult = await callGeminiWithBackoff(payload, 2);
        if (aiResult && aiResult.overallSummary) {
          finalGuidance = aiResult;
        }
      } catch (e) {
        // 网络不可用时自动平滑使用本地规则诊断结果
      }

      record.aiGuidance = finalGuidance;
      await syncEntityToCloud("records", record.id, record);

      document.getElementById("aiSummaryText").textContent = finalGuidance.overallSummary;
      document.getElementById("aiErrorTypeText").textContent = finalGuidance.errorTypeAnalysis;
      document.getElementById("aiDifficultyText").textContent = finalGuidance.difficultyDiagnosis;
      document.getElementById("aiActionAdviceText").textContent = finalGuidance.wordCategoryAnalysis;

      loadingEl.classList.add("hidden");
      contentEl.classList.remove("hidden");
      refreshAllPortalViews();
    }

    // 校验学生提交的拼写
    function handleSpellingSubmit(e) {
      if (e) e.preventDefault();
      if (gameState.phase !== "PLAYING") return;

      const input = document.getElementById("spellingInput");
      const rawValue = input.value;
      const normalized = normalizeSpelling(rawValue);

      // 防止误触空白回车直接判定坠机，给予轻晃提醒
      if (!normalized) {
        const card = document.getElementById("dictationCard");
        const toast = document.getElementById("emptyInputToast");
        card.classList.remove("animate-shake");
        void card.offsetWidth;
        card.classList.add("animate-shake");
        toast.classList.remove("hidden");
        setTimeout(() => toast.classList.add("hidden"), 1300);
        input.focus();
        return;
      }

      const currentItem = gameState.queue[gameState.currentIndex];
      const isCorrect = currentItem.accepts.some(ans => normalizeSpelling(ans) === normalized);

      if (isCorrect) {
        // ✅ 拼写正确：触发尾焰瞬间变大与高速爬升
        gameState.correctCount += 1;
        gameState.history.push({
          word: currentItem.word,
          cn: currentItem.cn,
          ipa: currentItem.ipa,
          userAnswer: rawValue.trim(),
          correct: true,
          reason: ""
        });

        gameState.phase = "BOOSTING";
        gameState.boostTimer = 0.85;
        gameState.targetProgress = gameState.correctCount / gameState.queue.length;
        gameState.flameScale = 2.55;
        gameState.cameraShake = 6;
        input.disabled = true;

        soundFX.playBoostThrust();
      } else {
        // ❌ 拼写错误：火箭直接失去动力俯冲坠毁
        triggerRocketCrash(rawValue.trim(), "拼写错误");
      }
    }

    // 触发坠机结束流程
    function triggerRocketCrash(userAnswerText, reasonLabel) {
      if (gameState.phase === "CRASHING" || gameState.phase === "END") return;

      const currentItem = gameState.queue[gameState.currentIndex];
      gameState.history.push({
        word: currentItem.word,
        cn: currentItem.cn,
        ipa: currentItem.ipa,
        userAnswer: userAnswerText || "（未作答 / 超时）",
        correct: false,
        reason: reasonLabel
      });

      gameState.phase = "CRASHING";
      gameState.crashPhase = "NOSEDIVE";
      gameState.crashTimer = 0;
      gameState.flameScale = 0;
      gameState.crashReason = reasonLabel;
      document.getElementById("spellingInput").disabled = true;
      pronouncer.stop();
    }

    function escapeHtml(str) {
      return String(str || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }

    // 计算火箭当前屏幕坐标 (从左下角发射台起飞，向右上角攀升)
    function getRocketPosition() {
      const isMobile = viewWidth < 768;

      // 发射台基准点：左下角
      const launchX = isMobile ? viewWidth * 0.22 : viewWidth * 0.14;
      const launchY = isMobile ? viewHeight * 0.50 : viewHeight * 0.82;

      // 爬升终点区域：保留在屏幕左侧/中上方，永不被右侧答题框遮挡
      const apexX = isMobile ? viewWidth * 0.56 : viewWidth * 0.42;
      const apexY = isMobile ? viewHeight * 0.16 : viewHeight * 0.22;

      const p = Math.max(0, Math.min(1, gameState.altitudeProgress));
      const baseX = launchX + (apexX - launchX) * p;
      const baseY = launchY + (apexY - launchY) * p;

      return {
        launchX,
        launchY,
        x: baseX,
        y: baseY + gameState.sinkOffsetPx
      };
    }

    // 颜色线性插值辅助函数
    function lerpColor(hexA, hexB, t) {
      const a = parseInt(hexA.slice(1), 16);
      const b = parseInt(hexB.slice(1), 16);
      const rA = (a >> 16) & 255, gA = (a >> 8) & 255, bA = a & 255;
      const rB = (b >> 16) & 255, gB = (b >> 8) & 255, bB = b & 255;
      const r = Math.round(rA + (rB - rA) * t);
      const g = Math.round(gA + (gB - gA) * t);
      const bl = Math.round(bA + (bB - bA) * t);
      return `rgb(${r}, ${g}, ${bl})`;
    }

    // 绘制随飞行高度动态过渡的宇宙大气背景
    function drawSkyAndSpace(alt, timeSec) {
      let topColor, midColor, botColor;

      if (alt < 0.35) {
        // 0 ~ 35%: 地球晴朗蓝天 -> 高空平流层深蓝
        const t = alt / 0.35;
        topColor = lerpColor("#1d4ed8", "#0f172a", t);
        midColor = lerpColor("#38bdf8", "#1e3a8a", t);
        botColor = lerpColor("#bae6fd", "#3b82f6", t);
      } else if (alt < 0.7) {
        // 35% ~ 70%: 平流层 -> 亚轨道深邃靛蓝
        const t = (alt - 0.35) / 0.35;
        topColor = lerpColor("#0f172a", "#050816", t);
        midColor = lerpColor("#1e3a8a", "#0f172a", t);
        botColor = lerpColor("#3b82f6", "#1e1b4b", t);
      } else {
        // 70% ~ 100%: 浩瀚深空星海
        const t = (alt - 0.7) / 0.3;
        topColor = "#03050c";
        midColor = lerpColor("#0f172a", "#090d1f", t);
        botColor = lerpColor("#1e1b4b", "#172554", t);
      }

      const grad = ctx.createLinearGradient(0, 0, 0, viewHeight);
      grad.addColorStop(0, topColor);
      grad.addColorStop(0.55, midColor);
      grad.addColorStop(1, botColor);
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, viewWidth, viewHeight);

      // 1. 星空层 (在开始界面与中高空清晰闪烁)
      const starVisibility =
        gameState.phase === "START"
          ? 0.65
          : Math.max(0, Math.min(1, (alt - 0.15) / 0.45));

      if (starVisibility > 0.02) {
        stars.forEach(s => {
          const twinkle = 0.55 + 0.45 * Math.sin(timeSec * s.twinkleSpeed + s.phase);
          ctx.globalAlpha = starVisibility * twinkle;
          ctx.fillStyle = s.tint;
          ctx.beginPath();
          ctx.arc(s.xRatio * viewWidth, s.yRatio * viewHeight, s.size, 0, Math.PI * 2);
          ctx.fill();
        });
        ctx.globalAlpha = 1;
      }

      // 2. 深空天体装饰 (高空阶段显现：土星环行星 & 弯月)
      if (alt > 0.45) {
        const spacePropAlpha = Math.min(0.9, (alt - 0.45) / 0.35);
        ctx.save();
        ctx.globalAlpha = spacePropAlpha;

        // 远景带光环行星
        const px = viewWidth * 0.33;
        const py = viewHeight * 0.16;
        const pGrad = ctx.createRadialGradient(px - 8, py - 8, 4, px, py, 28);
        pGrad.addColorStop(0, "#fde68a");
        pGrad.addColorStop(1, "#d97706");
        ctx.fillStyle = pGrad;
        ctx.beginPath();
        ctx.arc(px, py, 26, 0, Math.PI * 2);
        ctx.fill();

        // 行星光环
        ctx.strokeStyle = "rgba(253, 230, 138, 0.55)";
        ctx.lineWidth = 4;
        ctx.beginPath();
        ctx.ellipse(px, py, 48, 12, -0.28, 0, Math.PI * 2);
        ctx.stroke();

        ctx.restore();
      }

      // 3. 低空云朵层 (随着高度上升向下移出视野)
      const cloudAlpha = Math.max(0, 1 - alt * 2.3);
      if (cloudAlpha > 0.02) {
        ctx.save();
        ctx.globalAlpha = cloudAlpha * 0.82;
        ctx.fillStyle = "#ffffff";
        clouds.forEach(c => {
          const cx = ((c.xRatio + timeSec * c.speed) % 1.2 - 0.1) * viewWidth;
          const cy = c.yRatio * viewHeight + alt * viewHeight * 0.95;
          drawCloudShape(cx, cy, c.scale);
        });
        ctx.restore();
      }

      // 4. 地面发射场与左下角发射架 (随火箭升空向下平稳退出)
      const groundShiftY = alt * viewHeight * 1.15;
      if (groundShiftY < viewHeight * 0.65) {
        drawLaunchpadGround(groundShiftY);
      }
    }

    function drawCloudShape(x, y, s) {
      ctx.beginPath();
      ctx.arc(x, y, 26 * s, 0, Math.PI * 2);
      ctx.arc(x + 22 * s, y - 8 * s, 32 * s, 0, Math.PI * 2);
      ctx.arc(x + 48 * s, y, 24 * s, 0, Math.PI * 2);
      ctx.fill();
    }

    function drawLaunchpadGround(offsetY) {
      const pos = getRocketPosition();
      const groundY = pos.launchY + 54 + offsetY;

      ctx.save();
      // 远景丘陵剪影
      ctx.fillStyle = "#065f46";
      ctx.beginPath();
      ctx.moveTo(0, groundY + 12);
      ctx.quadraticCurveTo(viewWidth * 0.25, groundY - 26, viewWidth * 0.55, groundY + 8);
      ctx.quadraticCurveTo(viewWidth * 0.8, groundY - 18, viewWidth, groundY + 10);
      ctx.lineTo(viewWidth, viewHeight + 100);
      ctx.lineTo(0, viewHeight + 100);
      ctx.closePath();
      ctx.fill();

      // 地面草坪与发射场基座
      ctx.fillStyle = "#047857";
      ctx.fillRect(0, groundY + 6, viewWidth, viewHeight - groundY + 100);

      // 混凝土发射平台 (左下角)
      ctx.fillStyle = "#475569";
      ctx.beginPath();
      ctx.roundRect(pos.launchX - 58, groundY - 4, 116, 18, 6);
      ctx.fill();

      // 发射架铁塔桁架
      ctx.strokeStyle = "#94a3b8";
      ctx.lineWidth = 3;
      const towerX = pos.launchX - 38;
      ctx.strokeRect(towerX, groundY - 98, 16, 94);
      ctx.beginPath();
      ctx.moveTo(towerX, groundY - 98);
      ctx.lineTo(towerX + 16, groundY - 66);
      ctx.lineTo(towerX, groundY - 34);
      ctx.lineTo(towerX + 16, groundY - 4);
      ctx.stroke();

      // 顶部红色信标灯
      ctx.fillStyle = "#ef4444";
      ctx.beginPath();
      ctx.arc(towerX + 8, groundY - 103, 4, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
    }

    // 绘制精致矢量火箭与动态燃料尾焰
    function drawRocket(x, y, angle, flameScale, timeSec) {
      if (gameState.phase === "CRASHING" && gameState.crashPhase === "EXPLODED") {
        return; // 已爆炸解体后不再绘制完整箭体
      }

      ctx.save();
      ctx.translate(x, y);
      ctx.rotate(angle);

      // 1. 绘制尾部推进火焰 (随剩余时间缩小，答对瞬间暴涨)
      if (flameScale > 0.03) {
        const flicker = 0.88 + 0.24 * Math.sin(timeSec * 42);
        const flameLen = (48 * flameScale) * flicker;
        const flameWidth = 17 * Math.pow(flameScale, 0.65);

        // 外层炽热橙红光晕
        const outerGrad = ctx.createLinearGradient(0, 34, 0, 34 + flameLen * 1.25);
        outerGrad.addColorStop(0, "rgba(251, 146, 60, 0.95)");
        outerGrad.addColorStop(0.55, "rgba(239, 68, 68, 0.7)");
        outerGrad.addColorStop(1, "rgba(239, 68, 68, 0)");

        ctx.fillStyle = outerGrad;
        ctx.beginPath();
        ctx.moveTo(-flameWidth, 34);
        ctx.quadraticCurveTo(0, 34 + flameLen * 1.35, flameWidth, 34);
        ctx.closePath();
        ctx.fill();

        // 内层白金/淡蓝高温核心焰芯
        const innerGrad = ctx.createLinearGradient(0, 34, 0, 34 + flameLen * 0.75);
        innerGrad.addColorStop(0, "#ffffff");
        innerGrad.addColorStop(0.5, "#fef08a");
        innerGrad.addColorStop(1, "rgba(56, 189, 248, 0)");

        ctx.fillStyle = innerGrad;
        ctx.beginPath();
        ctx.moveTo(-flameWidth * 0.55, 34);
        ctx.quadraticCurveTo(0, 34 + flameLen * 0.85, flameWidth * 0.55, 34);
        ctx.closePath();
        ctx.fill();
      }

      // 2. 两侧空气动力学尾翼
      ctx.fillStyle = "#e11d48";
      // 左尾翼
      ctx.beginPath();
      ctx.moveTo(-16, 10);
      ctx.lineTo(-32, 34);
      ctx.lineTo(-14, 30);
      ctx.closePath();
      ctx.fill();
      // 右尾翼
      ctx.beginPath();
      ctx.moveTo(16, 10);
      ctx.lineTo(32, 34);
      ctx.lineTo(14, 30);
      ctx.closePath();
      ctx.fill();

      // 3. 底部喷口
      ctx.fillStyle = "#334155";
      ctx.beginPath();
      ctx.moveTo(-11, 28);
      ctx.lineTo(11, 28);
      ctx.lineTo(14, 36);
      ctx.lineTo(-14, 36);
      ctx.closePath();
      ctx.fill();

      // 4. 火箭主箭体 (流线型金属白渐变)
      const bodyGrad = ctx.createLinearGradient(-18, 0, 18, 0);
      bodyGrad.addColorStop(0, "#e2e8f0");
      bodyGrad.addColorStop(0.45, "#ffffff");
      bodyGrad.addColorStop(1, "#cbd5e1");

      ctx.fillStyle = bodyGrad;
      ctx.beginPath();
      ctx.moveTo(0, -48);
      ctx.bezierCurveTo(18, -26, 18, 8, 15, 29);
      ctx.lineTo(-15, 29);
      ctx.bezierCurveTo(-18, 8, -18, -26, 0, -48);
      ctx.closePath();
      ctx.fill();

      // 5. 红色整流罩鼻锥
      ctx.fillStyle = "#e11d48";
      ctx.beginPath();
      ctx.moveTo(0, -48);
      ctx.bezierCurveTo(10, -36, 13, -27, 14, -22);
      ctx.lineTo(-14, -22);
      ctx.bezierCurveTo(-13, -27, -10, -36, 0, -48);
      ctx.closePath();
      ctx.fill();

      // 6. 太空舷窗
      ctx.fillStyle = "#1e293b";
      ctx.beginPath();
      ctx.arc(0, -4, 9.5, 0, Math.PI * 2);
      ctx.fill();

      const windowGrad = ctx.createRadialGradient(-2, -6, 1, 0, -4, 7.5);
      windowGrad.addColorStop(0, "#bae6fd");
      windowGrad.addColorStop(1, "#0284c7");
      ctx.fillStyle = windowGrad;
      ctx.beginPath();
      ctx.arc(0, -4, 7, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
    }

    // 更新并绘制粒子与爆炸冲击波
    function updateAndDrawParticles(dt) {
      // 冲击波光环
      for (let i = shockwaves.length - 1; i >= 0; i--) {
        const sw = shockwaves[i];
        sw.radius += (sw.maxRadius - sw.radius) * Math.min(1, dt * 6.5);
        sw.alpha -= dt * 1.4;
        if (sw.alpha <= 0) {
          shockwaves.splice(i, 1);
          continue;
        }
        ctx.save();
        ctx.strokeStyle = `rgba(253, 224, 71, ${sw.alpha})`;
        ctx.lineWidth = 5 * sw.alpha;
        ctx.beginPath();
        ctx.arc(sw.x, sw.y, sw.radius, 0, Math.PI * 2);
        ctx.stroke();
        ctx.restore();
      }

      // 粒子更新与渲染
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        if (p.type === "debris") {
          p.vy += 260 * dt; // 碎片受重力下坠
        }
        p.alpha -= p.decay * dt;

        if (p.alpha <= 0) {
          particles.splice(i, 1);
          continue;
        }

        ctx.save();
        ctx.globalAlpha = Math.max(0, p.alpha);
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }
    }

    // 更新 HUD 倒计时进度条与数字色彩
    function updateTimerUI() {

      num.textContent = `${Math.max(0, gameState.timeLeft).toFixed(1)}s`;

      if (ratio > 0.45) {

        num.className = "font-mono text-sm font-extrabold text-emerald-600";

      } else if (ratio > 0.22) {
        bar.className = "h-full rounded-full bg-amber-500 transition-colors duration-200";

        label.innerHTML = isEn ? `<span>⚠️ Low Fuel — Rocket Descending!</span>` : `<span>⚠️ 燃料下降中，火箭正在下沉！</span>`;

      }
    }

    // 主动画循环
    let lastTimestamp = 0;
    function mainLoop(timestamp) {
      if (!lastTimestamp) lastTimestamp = timestamp;
      const dt = Math.min((timestamp - lastTimestamp) / 1000, 0.05);
      lastTimestamp = timestamp;
      const timeSec = timestamp / 1000;

      // 1. 逻辑步进更新
      if (gameState.phase === "START") {
        // 开始界面：火箭在左下角发射台点火怠速预热
        gameState.flameScale = 0.55 + 0.08 * Math.sin(timeSec * 6);
        gameState.sinkOffsetPx = 0;
        gameState.rocketAngle = 0.36;
      } else if (gameState.phase === "PLAYING") {
        // 拼写倒计时中：火箭慢慢下降，尾部火焰逐渐变小
        gameState.timeLeft -= dt;
        const fuelRatio = Math.max(0, gameState.timeLeft / gameState.timeLimit);

        // 下沉距离随时间流逝逐步增大 (最多下沉约 95px)
        const targetSink = (1 - fuelRatio) * 95;
        gameState.sinkOffsetPx += (targetSink - gameState.sinkOffsetPx) * Math.min(1, dt * 8);

        // 尾焰随时间流逝从 1.05 逐渐缩小到 0.14
        gameState.flameScale = 0.14 + fuelRatio * 0.92;

        // 低燃料最后 3 秒声光提醒
        const ceilSec = Math.ceil(gameState.timeLeft);
        if (ceilSec <= 3 && ceilSec >= 1 && ceilSec < gameState.lastWarnSecond) {
          gameState.lastWarnSecond = ceilSec;
          soundFX.playLowFuelTick();
        }

        // 持续喷出少量尾烟粒子
        const pos = getRocketPosition();
        if (Math.random() < 0.45) {
          spawnThrustParticles(pos.x - 10, pos.y + 28, gameState.rocketAngle, 1, fuelRatio * 0.7);
        }

        updateTimerUI();

        // 时限归零 -> 直接坠机
        if (gameState.timeLeft <= 0) {
          gameState.timeLeft = 0;
          updateTimerUI();
          triggerRocketCrash("", "思考超时");
        }
      } else if (gameState.phase === "BOOSTING") {
        // 拼写正确：尾焰一瞬间变大，快速上升一大段距离
        gameState.boostTimer -= dt;
        gameState.altitudeProgress +=
          (gameState.targetProgress - gameState.altitudeProgress) * Math.min(1, dt * 7.5);
        gameState.sinkOffsetPx += (0 - gameState.sinkOffsetPx) * Math.min(1, dt * 10);
        gameState.flameScale = 1.3 + (gameState.boostTimer / 0.85) * 1.45;
        gameState.cameraShake = Math.max(0, gameState.cameraShake - dt * 10);

        const pos = getRocketPosition();
        spawnThrustParticles(pos.x - 12, pos.y + 30, gameState.rocketAngle, 4, 1.5);

        if (gameState.boostTimer <= 0) {
          gameState.altitudeProgress = gameState.targetProgress;
          gameState.currentIndex += 1;
          if (gameState.currentIndex >= gameState.queue.length) {
            // 全部 19 词通关！
            showEndScreen(true);
          } else {
            loadCurrentWord();
          }
        }
      } else if (gameState.phase === "CRASHING") {
        // 拼写错误或超时：俯冲坠机并爆炸
        gameState.crashTimer += dt;
        const pos = getRocketPosition();

        if (gameState.crashPhase === "NOSEDIVE") {
          // 0 ~ 0.55s: 失去动力，机头下栽加速下坠
          gameState.rocketAngle += dt * 3.8;
          gameState.sinkOffsetPx += dt * 340;
          if (Math.random() < 0.7) {
            particles.push({
              x: pos.x,
              y: pos.y,
              vx: (Math.random() - 0.5) * 30,
              vy: -35,
              size: 8 + Math.random() * 7,
              alpha: 0.75,
              decay: 1.1,
              color: "#475569",
              type: "smoke"
            });
          }

          if (gameState.crashTimer >= 0.52 || pos.y >= viewHeight - 35) {
            gameState.crashPhase = "EXPLODED";
            gameState.cameraShake = 18;
            soundFX.playCrashExplosion();
            triggerCrashExplosionParticles(pos.x, pos.y);
          }
        } else {
          // 爆炸后等待 1.15s 展示火光烟雾，再弹出结算报告
          gameState.cameraShake = Math.max(0, gameState.cameraShake - dt * 22);
          if (gameState.crashTimer >= 1.75) {
            showEndScreen(false);
          }
        }
      }

      // 2. Canvas 画面渲染
      ctx.save();
      if (gameState.cameraShake > 0.1) {
        const sx = (Math.random() - 0.5) * gameState.cameraShake;
        const sy = (Math.random() - 0.5) * gameState.cameraShake;
        ctx.translate(sx, sy);
      }

      drawSkyAndSpace(gameState.altitudeProgress, timeSec);
      updateAndDrawParticles(dt);

      const rocketPos = getRocketPosition();
      drawRocket(
        rocketPos.x,
        rocketPos.y,
        gameState.rocketAngle,
        gameState.flameScale,
        timeSec
      );

      ctx.restore();
      requestAnimationFrame(mainLoop);
    }

    // 渲染选词与可编辑词汇表格

      const container = document.getElementById("vocabPreviewList");
      const query = vocabManagerState.searchQuery.trim().toLowerCase();
      container.innerHTML = "";

      const filtered = vocabManagerState.items
        .map((item, index) => ({ item, index }))
        .filter(({ item }) => {
          if (!query) return true;

            item.word.toLowerCase().includes(query) ||

      container.querySelectorAll(".vocab-row-check").forEach(cb => {

    // 导出当前词表为 UTF-8 BOM CSV 文件
    function exportCurrentVocabCsv() {
      const header = ["Word", "Phonetic", "Meaning", "Selected"];
      const lines = [header.join(",")];
      vocabManagerState.items.forEach(item => {
        const row = [
          `"${item.word.replace(/"/g, '""')}"`,
          `"${item.ipa.replace(/"/g, '""')}"`,
          `"${item.cn.replace(/"/g, '""')}"`,
          item.selected ? "1" : "0"
        ];
        lines.push(row.join(","));
      });
      const blob = new Blob(["\uFEFF" + lines.join("\n")], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `${vocabManagerState.currentBankName || "词汇表"}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

      // 文档上传与拖拽事件
      const docInput = document.getElementById("vocabDocInput");
      const docDropZone = document.getElementById("docDropZone");
      docInput.addEventListener("change", (e) => {
        if (e.target.files && e.target.files[0]) {
          handleDocumentFileUpload(e.target.files[0]);
          e.target.value = "";
        }
      });
      docDropZone.addEventListener("dragover", (e) => {
        e.preventDefault();
        docDropZone.classList.add("border-sky-600", "bg-sky-50");
      });
      docDropZone.addEventListener("dragleave", () => {
        docDropZone.classList.remove("border-sky-600", "bg-sky-50");
      });
      docDropZone.addEventListener("drop", (e) => {
        e.preventDefault();
        docDropZone.classList.remove("border-sky-600", "bg-sky-50");
        if (e.dataTransfer.files && e.dataTransfer.files[0]) {
          handleDocumentFileUpload(e.dataTransfer.files[0]);
        }
      });

      // 粘贴文本解析（本地正则 & AI 智能整理）
      document.getElementById("parsePasteLocalBtn").addEventListener("click", () => {
        const raw = document.getElementById("pasteVocabTextarea").value;
        const items = parseVocabTextLocally(raw);
        applyImportedItems(items, "粘贴文本词表");
      });
      document.getElementById("parsePasteAiBtn").addEventListener("click", async () => {
        const raw = document.getElementById("pasteVocabTextarea").value.trim();
        if (!raw) {
          setImportStatus("error", "⚠️ 请先在输入框中粘贴单词文本。", false);
          return;
        }
        setImportStatus("loading", "✨ AI 正在智能整理并补全音标与中文释义...", true);
        try {
          const items = await extractVocabFromTextWithAI(raw);
          applyImportedItems(items, "AI整理词表");
        } catch (err) {
          setImportStatus("error", `❌ AI 整理失败：${err.message || "请重试"}`, false);
        }
      });

      // 全选 / 取消全选 / 反选 / 新增一词 / AI 补全 / 搜索过滤
      document.getElementById("selectAllWordsBtn").addEventListener("click", () => {
        vocabManagerState.items.forEach(it => { it.selected = true; });
        renderVocabManagerTable();
      });
      document.getElementById("clearAllWordsBtn").addEventListener("click", () => {
        vocabManagerState.items.forEach(it => { it.selected = false; });
        renderVocabManagerTable();
      });
      document.getElementById("invertWordsBtn").addEventListener("click", () => {
        vocabManagerState.items.forEach(it => { it.selected = !it.selected; });
        renderVocabManagerTable();
      });
      document.getElementById("addManualRowBtn").addEventListener("click", () => {
        vocabManagerState.items.unshift(buildNormalizedVocabItem("new word", "", "", [], true));
        renderVocabManagerTable();
      });
      document.getElementById("aiFillMissingBtn").addEventListener("click", fillMissingVocabFieldsWithAI);
      document.getElementById("vocabSearchInput").addEventListener("input", (e) => {
        vocabManagerState.searchQuery = e.target.value;
        renderVocabManagerTable();
      });

      // 保存自定义词库 / 导出 CSV / 恢复默认 / 删除自建词库
      document.getElementById("toggleSaveBankBtn").addEventListener("click", () => {
        const bar = document.getElementById("saveBankBar");
        bar.classList.toggle("hidden");
        if (!bar.classList.contains("hidden")) {
          const inp = document.getElementById("customBankNameInput");
          inp.value = vocabManagerState.currentBankId === "default" ? "" : vocabManagerState.currentBankName;
          inp.focus();
        }
      });
      document.getElementById("cancelSaveBankBtn").addEventListener("click", () => {
        document.getElementById("saveBankBar").classList.add("hidden");
      });
      document.getElementById("confirmSaveBankBtn").addEventListener("click", async () => {
        const nameInput = document.getElementById("customBankNameInput");
        const bankName = nameInput.value.trim();
        if (!bankName) {
          setImportStatus("error", "⚠️ 请输入词库名称后再保存。", false);
          nameInput.focus();
          return;
        }
        const validItems = vocabManagerState.items
          .filter(it => it.word.trim())
          .map(it => ({ ...it, accepts: [...(it.accepts || [it.word])] }));
        if (validItems.length === 0) {
          setImportStatus("error", "⚠️ 当前词表为空，无法保存。", false);
          return;
        }

        const bankId =
          vocabManagerState.currentBankId &&
          vocabManagerState.currentBankId !== "default" &&
          vocabManagerState.currentBankId !== "unsaved_custom"
            ? vocabManagerState.currentBankId
            : `bank_${Date.now()}`;

        const bankObj = {
          id: bankId,
          name: bankName,
          items: validItems,
          updatedAt: Date.now()
        };

        const existingIdx = portalDataStore.customBanks.findIndex(b => b.id === bankId);
        if (existingIdx >= 0) {
          portalDataStore.customBanks[existingIdx] = bankObj;
        } else {
          portalDataStore.customBanks.unshift(bankObj);
        }
        await syncEntityToCloud("vocab_banks", bankId, bankObj);

        vocabManagerState.currentBankId = bankId;
        vocabManagerState.currentBankName = bankName;
        document.getElementById("saveBankBar").classList.add("hidden");
        syncActiveBankBanner();
        if (portalDataStore.currentRole === "PARENT") renderParentPortal();
        setImportStatus("success", `💾 词库「${bankName}」已成功保存！`, false);
      });

      document.getElementById("exportCsvBtn").addEventListener("click", exportCurrentVocabCsv);

      document.getElementById("resetDefaultVocabBtn").addEventListener("click", () => {
        vocabManagerState.currentBankId = "default";
        vocabManagerState.currentBankName = "太空单元默认词库";
        vocabManagerState.items = DEFAULT_VOCABULARY_LIST.map(item => ({
          ...item,
          accepts: [...item.accepts],
          selected: true
        }));
        renderVocabManagerTable();
        setImportStatus("success", "🔄 已恢复为默认太空单元 19 词词表。", false);
      });

      document.getElementById("deleteSavedBankBtn").addEventListener("click", async () => {
        const deletedId = vocabManagerState.currentBankId;
        const savedBanks = loadSavedCustomBanks().filter(b => b.id !== deletedId);
        persistSavedCustomBanks(savedBanks);
        await deleteEntityFromCloud("vocab_banks", deletedId);

        vocabManagerState.currentBankId = "default";
        vocabManagerState.currentBankName = "太空单元默认词库";
        vocabManagerState.items = DEFAULT_VOCABULARY_LIST.map(item => ({
          ...item,
          accepts: [...item.accepts],
          selected: true
        }));
        renderVocabManagerTable();
        if (portalDataStore.currentRole === "PARENT") {
          renderParentPortal();
        }
        setImportStatus("success", "🗑️ 已删除该自建词库并切换回默认词库。", false);
      });

      // 6. 游戏中重听按钮 & 全局 Tab 键快捷重听
      document.getElementById("replayAudioBtn").addEventListener("click", () => {
        if (gameState.phase === "PLAYING") {
          pronouncer.speak(gameState.queue[gameState.currentIndex].word);
          document.getElementById("spellingInput").focus();
        }
      });

      window.addEventListener("keydown", (e) => {
        if (e.key === "Tab" && gameState.phase === "PLAYING") {
          e.preventDefault();
          pronouncer.speak(gameState.queue[gameState.currentIndex].word);
          document.getElementById("spellingInput").focus();
        }
      });

      // 7. 表单提交拼写检验
      document.getElementById("dictationForm").addEventListener("submit", handleSpellingSubmit);

      // 8. 中途返回与结算界面按钮
      document.getElementById("abortGameBtn").addEventListener("click", returnToStartMenu);
      document.getElementById("backToMenuBtn").addEventListener("click", returnToStartMenu);
      document.getElementById("restartSameBtn").addEventListener("click", () => {
        startGame(gameState.difficulty, {
          customWords: gameState.queue,
          taskId: portalDataStore.activeTaskId,
          customTimeLimit: gameState.timeLimit
        });
      });

      // 启动 Canvas 60fps 动画循环
      requestAnimationFrame(mainLoop);
    };
  </script>
</body>
